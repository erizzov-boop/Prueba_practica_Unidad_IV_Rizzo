% Diagrama de actividades - Registrar pedido
\begin{tikzpicture}[
    node distance=1.1cm and 2.2cm,
    initial/.style={circle, draw=black, fill=black, minimum size=0.35cm},
    final/.style={circle, draw=black, fill=white, minimum size=0.45cm, double, double distance=1.3pt},
    action/.style={rectangle, rounded corners=5pt, draw=black, fill=white, text=black,
        minimum width=3.2cm, minimum height=0.9cm, align=center, font=\small},
    decision/.style={diamond, draw=black, fill=white, text=black, aspect=2,
        inner sep=1pt, align=center, font=\scriptsize},
    every path/.style={draw=black, thick, -{Latex[length=2.5mm]}},
    lbl/.style={font=\scriptsize, text=black}
]

\node[initial] (start) {};
\node[action, below=of start] (recibir) {Recibir pedido};
\node[decision, below=1.3cm of recibir] (stock) {¿Stock\\disponible?};
\node[action, below left=1.4cm and 0.3cm of stock] (confirmar) {Confirmar pedido};
\node[action, below right=1.4cm and 0.3cm of stock] (notificar) {Notificar faltante};
\node[final, below=2.6cm of stock] (end) {};

\draw (start) -- (recibir);
\draw (recibir) -- (stock);
\draw (stock) -| (confirmar) node[lbl, pos=0.25, above] {Sí};
\draw (stock) -| (notificar) node[lbl, pos=0.25, above] {No};
\draw (confirmar) |- (end);
\draw (notificar) |- (end);

\end{tikzpicture}
